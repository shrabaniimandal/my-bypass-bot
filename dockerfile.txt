FROM python:3.9

# Java এবং Apktool ইন্সটল করা
RUN apt-get update && apt-get install -y openjdk-17-jre-headless wget

# Apktool এবং Signer ডাউনলোড করা
RUN wget https://raw.githubusercontent.com/iBotPehenlo/Apktool/master/apktool && \
    chmod +x apktool && \
    mv apktool /usr/local/bin/apktool

RUN wget https://github.com/kevinalbrawan/Uber-APK-Signer/releases/download/v0.2.4/uber-apk-signer.jar -O /uber-apk-signer.jar

WORKDIR /app
COPY . .
RUN pip install -r requirements.txt

CMD ["python", "bot.py"]
